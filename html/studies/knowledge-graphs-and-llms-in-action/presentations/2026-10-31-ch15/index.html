def get_prompt(self, name, **kwargs):
    system = self.config["prompts"][name].get("system")
    template_file = self.base / self.config["prompts"][name]["template"]
    prompt = jinja2_formatter(template_file.read_text(), **kwargs)
    return system, prompt

def getAnnotations(self, reload=True):
    if reload:
        self.config = self.load()

템플릿 파일은 호출마다 다시 읽히고, 노트·예시는 getAnnotations가 YAML을 다시 읽는다(AIML Quant 보충).

def get_schema(self):                       # 표본 없이 DB 전체
    result = list(session.run("CALL apoc.meta.schema({sample:-1})"))[0]["value"]
    [Node.mk_node(k, v) for k, v in result.items() if v["type"] == "node"]

def apply_configuration(config):
    Node.items = {n.name: n for n in Node.items.values()
                  if n.name not in config["skip"]["classes"]}               # ① 노드
    for node in Node.items.values():
        node.drop_properties(config["skip"]["properties"])                    # ② 속성
        node.description = config["descriptions"]["classes"].get(node.name)  # ③ 설명

(설명용 축약) ④ 관계는 출발·도착 노드나 이름이 스킵 대상이면 뺀다. 설명(③)은 노드에만 붙는다.

결과 텍스트가 질의 생성 프롬프트의 스키마 자리에 들어간다. 관계 설명이 필요하면 이 부분을 직접 보완해야 한다.

15.1.4 상태 관리 설계 — 에이전트들이 함께 쓰는 공유 메모리

각 에이전트가 정해진 필드를 채우므로 틀린 단계를 필드 이름으로 거슬러 올라갈 수 있다.

class AgentState(TypedDict):
    question: str                                          # 질문 입력
    output_type: str; output_type_reason: str              # 의도 감지 결과
    schema: str                                            # 스키마 정보
    query: str; query_reasoning: str; query_message: str   # 질의 생성
    results_error: list                                    # 오류 처리
    summary: str; summary_reason: str; summary_analysis: bool  # 요약 생성
    information: str; retries: int                         # 질의 재시도

마지막 줄의 information·retries는 결과물이 아니라 재시도 판정과 오류 맥락을 위한 흐름 제어용이다.

저자들은 이것을 책임의 사슬이라 부른다. 상태는 데이터와 함께 라우팅·오류 처리에 필요한 맥락도 들고 다닌다.

15.1.5 파이프라인 에이전트 구현 — 단계마다 전문 에이전트 하나

다섯 에이전트는 모두 AgentState를 받아 자기가 채울 필드만 딕셔너리로 돌려준다.

윗줄은 실행 순서, 아래 상자는 각 에이전트가 반환하는 필드다. 주황색은 선언과 다른 반환 이름, 점선은 조건이 맞을 때만 지나는 경로다.

LLM을 부르는 세 에이전트는 설정에서 프롬프트를 받아 공통 함수 run_prompt로 실행한다.

def intent_detection(self, state: AgentState):
    system, prompt = self.config.get_prompt("intent_detection", question=state["question"])
    results = self.run_prompt(prompt, system)
    return {"output_type": results["type"], "output_reason": results["reason"]}

Listing 15.7은 원서에 이미지로만 실려 교재 저장소 코드로 옮겼다.

def schema_extraction(self, state: AgentState):          # Listing 15.7
    self.neo4j_schema.get_schema()
    self.neo4j_schema.apply_configuration()
    return {"schema": str(self.neo4j_schema), "retries": 0}
def text_to_cypher(self, state: AgentState):             # Listing 15.8
    extra = {"annotations": self.config.getAnnotations(), "selection": self.selection}
    params = dict(state) | extra
    system, prompt = self.config.get_prompt("text_to_cypher", **params)
    results = self.run_prompt(prompt, system)

dict(state) | extra: 상태 전체에 주석과 화면 선택을 더한 딕셔너리가 그대로 템플릿 변수가 된다.

재시도 때는 이 경로로 information(시도한 질의와 오류)이 질문 뒤에 붙는다. 템플릿이 요구하는 success 필드는 읽지 않는다.

실패하면 시도한 질의와 오류를 다음 시도의 재료로 남긴다

성공하면 그래프·지도는 레코드 그대로, 표는 DataFrame으로 결과를 둔다.

def query_execution(self, state: AgentState):
    try:
        results = self.neo4j_schema.run(state["query"])   # 형식에 맞춰 self.results에 보관
        results_error, information = None, ""
    except neo4j.exceptions.ClientError as e:
        results_error = str(e)
        information = f"We tried: {state['query']} and we got: {e}"
    retries = state.get("retries", 0) + 1
    return {"results_error": results_error, "retries": retries, "information": information}

(설명용 축약 — self.results 대입 생략) retries 줄은 try 밖에 있어 성공한 실행도 센다. 스키마 추출이 0으로 맞춰 두므로 첫 실행 뒤 1이 된다.

ClientError만 잡는다. 노트북은 LLM이 빈 질의를 내면 다른 예외로 파이프라인이 멈춘다고 기록했다.

def generate_summary(self, state: AgentState):
    extra = {"records": self.results, "selection": self.selection}
    params = dict(state) | extra
    system, prompt = self.config.get_prompt("generate_summary", **params)

    # display_message: 키가 있으면 그린다 — query_reasoning · table · map · query · summary

def route_after_query(state):
    if state.get('stop_reason') == 'deadline':
        return END                                   # 질문 전체 120초 초과
    if state.get('error'):
        return 'text_to_cypher' if state['retries'] < 3 else END   # 책: 오류면 재시도
    if state.get('feedback_kind') == 'broaden' and state['retries'] < 3:
        return 'text_to_cypher'                      # 추가: 0행이면 포함 조건으로

적재 뒤 데이터베이스를 읽기 전용으로 다시 띄운다. CREATE를 직접 보내 거부되는 것을 확인했고, 앱은 읽기 전용이 아니면 기동하지 않는다.

DB 질의 10초, LLM 요청 30초, 질문 전체 120초, 동시에 한 질문. 오류 메시지는 코드와 정해진 문구로만 보인다.

책의 질의 실행 에이전트는 ClientError만 잡았다. 여기서는 실패의 종류마다 갈 곳을 정했다.

"이 ETF"가 무엇인지는 화면의 선택에서 읽는다

§15.3에서 "선택한 범죄"를 현재 선택으로 풀던 것과 같다. 한 실행에서 생성된 질의:

MATCH (selected:ETF {code: '471760'})-[:FOCUSES_ON]->(sector:Sector)<-[:FOCUSES_ON]-(etf:ETF)
WHERE etf.code <> selected.code
  AND EXISTS {
    MATCH (selected)-[:FOCUSES_ON]->(technology:Technology)<-[:FOCUSES_ON]-(etf)
  }
RETURN DISTINCT etf.code AS code, etf.name AS name,
       etf.total_fee AS total_fee, etf.management_company AS management_company
ORDER BY total_fee ASC, code ASC
LIMIT 200

섹터와 기술을 모두 공유해야 한다는 조건을 EXISTS 부분 질의로 표현하고, 선택한 ETF 자신은 뺐다. 결과: 407160(0.445%), 479620(0.45%), 480310(0.49%), 공통 섹터 「반도체」·기술 「AI」.

출력 형식이 표라서 책의 라우팅대로 요약 없이 끝난다. 답은 표 자체다.
